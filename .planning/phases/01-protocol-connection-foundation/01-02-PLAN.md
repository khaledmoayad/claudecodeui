Create the `ccud/` directory structure with all daemon source files. All files are plain JavaScript ES Modules. Follow project conventions: 2-space indent, single quotes, semicolons, trailing commas.

1. **ccud/src/logger.js** -- stderr-only logging. CRITICAL: NEVER use console.log (it writes to stdout and corrupts JSON-RPC).
   ```javascript
   /**
    * @module ccud/logger
    * Daemon logging -- stderr only. stdout is reserved for JSON-RPC.
    */

   export function log(...args) {
     process.stderr.write(`[ccud] ${args.join(' ')}\n`);
   }

   export function error(...args) {
     process.stderr.write(`[ccud:error] ${args.join(' ')}\n`);
   }
   ```

2. **ccud/src/pid.js** -- PID file management at `~/.ccud/ccud.pid`. Set `process.title = 'ccud'` for ps visibility.
   ```javascript
   /**
    * @module ccud/pid
    * PID file management and orphan cleanup.
    */
   import { writeFileSync, unlinkSync, mkdirSync } from 'fs';
   import { join } from 'path';
   import { homedir } from 'os';
   import { log, error } from './logger.js';

   const PID_DIR = join(homedir(), '.ccud');
   const PID_FILE = join(PID_DIR, 'ccud.pid');

   export function writePidFile() {
     try {
       mkdirSync(PID_DIR, { recursive: true });
       writeFileSync(PID_FILE, String(process.pid), 'utf8');
       process.title = 'ccud';
       log(`PID ${process.pid} written to ${PID_FILE}`);
     } catch (e) {
       error(`Failed to write PID file: ${e.message}`);
     }
   }

   export function removePidFile() {
     try {
       unlinkSync(PID_FILE);
       log('PID file removed');
     } catch (e) {
       // File may already be gone -- not an error
     }
   }
   ```

3. **ccud/src/transport.js** -- Stdin/stdout JSON-RPC transport using readline. Newline-delimited messages. stdin close triggers graceful shutdown.
   ```javascript
   /**
    * @module ccud/transport
    * Stdin/stdout JSON-RPC transport with newline framing.
    */
   import { createInterface } from 'readline';
   import { log, error as logError } from './logger.js';

   export function createStdioTransport(onMessage) {
     const rl = createInterface({ input: process.stdin, crlfDelay: Infinity });

     rl.on('line', (line) => {
       if (!line.trim()) return;
       try {
         const parsed = JSON.parse(line);
         onMessage(parsed);
       } catch (e) {
         logError(`Invalid JSON-RPC: ${e.message}`);
       }
     });

     // CRITICAL: stdin close = SSH disconnected = self-terminate
     rl.on('close', () => {
       log('stdin closed, shutting down');
       // Grace period for pending operations
       setTimeout(() => process.exit(0), 5000);
     });

     return {
       send(msg) {
         const payload = Array.isArray(msg) ? JSON.stringify(msg) : JSON.stringify(msg);
         process.stdout.write(payload + '\n');
       },
     };
   }
   ```

4. **ccud/src/handlers/initialize.js** -- Handles the `initialize` JSON-RPC method. Returns daemon version, protocol version, node version, platform, arch.
   ```javascript
   /**
    * @module ccud/handlers/initialize
    * Initialize handshake handler.
    */
   import { log } from '../logger.js';

   const DAEMON_VERSION = '1.0.0';
   const PROTOCOL_VERSION = 1;

   export function handleInitialize(params) {
     log(`Initialize request from client v${params?.clientVersion || 'unknown'}, protocol v${params?.protocolVersion || 'unknown'}`);
     return {
       protocolVersion: PROTOCOL_VERSION,
       daemonVersion: DAEMON_VERSION,
       nodeVersion: process.version,
       platform: process.platform,
       arch: process.arch,
     };
   }

   export { DAEMON_VERSION, PROTOCOL_VERSION };
   ```

5. **ccud/src/handlers/fs.js** -- Filesystem operation stubs. Returns method-not-found for now (implemented in Phase 2).
   ```javascript
   /**
    * @module ccud/handlers/fs
    * Filesystem operation handlers (stubs for Phase 1).
    */
   export function handleFs(method, params) {
     return { error: { code: -32601, message: `Method not implemented: ${method}` } };
   }
   ```

6. **ccud/src/index.js** -- Main daemon entry point. Wires everything together:
   - Import all modules (logger, pid, transport, handlers)
   - Call `writePidFile()` immediately
   - Create transport via `createStdioTransport(handleIncoming)`
   - `handleIncoming(msg)` dispatches messages:
     - If `Array.isArray(msg)` -- batch request: `Promise.all(msg.map(processSingleMessage))`, filter null results (notifications), send array response
     - Otherwise call `processSingleMessage(msg)`, send response if not null
   - `processSingleMessage(msg)` function:
     - If msg has no `method` property, ignore (it's a response, not a request)
     - If msg has no `id` property, it's a notification -- process but return null (no response)
     - Route by `msg.method`:
       - `'initialize'` -> `handleInitialize(msg.params)` -> wrap in JSON-RPC success response `{ jsonrpc: '2.0', id: msg.id, result: ... }`
       - Methods starting with `'fs/'` -> `handleFs(msg.method, msg.params)` -- if result has `.error`, wrap in JSON-RPC error response; else wrap in success response
       - Unknown method -> JSON-RPC error response: `{ jsonrpc: '2.0', id: msg.id, error: { code: -32601, message: 'Method not found' } }`
   - Signal handlers: `process.on('SIGTERM', cleanup)`, `process.on('SIGINT', cleanup)`, `process.on('exit', cleanup)`
   - `cleanup()` function: call `removePidFile()`, attempt to kill process group with `process.kill(-process.pid, 'SIGTERM')` wrapped in try/catch (may not be group leader)
   - Log startup message to stderr: `log('Daemon started, PID ' + process.pid)`

All imports (fs, readline, os, path, crypto) are Node.js builtins. `jsonrpc-lite` is NOT used in the daemon (we hand-build the simple JSON objects since the daemon only needs to parse/stringify, and avoiding an external dep means zero bundling complexity). `chokidar` will be added in Phase 3 for file watching.

3. Create `ccud/build.js` -- esbuild bundling script:
   ```javascript
   import esbuild from 'esbuild';
   import { readFileSync } from 'fs';

   const pkg = JSON.parse(readFileSync(new URL('./package.json', import.meta.url), 'utf8'));

   await esbuild.build({
     entryPoints: ['ccud/src/index.js'],
     bundle: true,
     platform: 'node',
     target: 'node20',
     format: 'esm',
     outfile: 'dist/ccud.mjs',
     banner: { js: '#!/usr/bin/env node' },
     minify: false, // Keep readable for debugging on remote hosts
     define: {
       'CCUD_VERSION': JSON.stringify(pkg.version),
     },
   });

   console.error('Daemon bundled to dist/ccud.mjs');
   ```
   Note: `console.error` not `console.log` in the build script to model good habits, though this is build-time only.
