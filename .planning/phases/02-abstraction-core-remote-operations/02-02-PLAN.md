    function permToRwx(perm) {
      const r = perm & 4 ? 'r' : '-';
      const w = perm & 2 ? 'w' : '-';
      const x = perm & 1 ? 'x' : '-';
      return r + w + x;
    }

    - `fs/readFile`: `const content = await readFile(params.path, 'utf8'); return { content };`
    - `fs/writeFile`: `await writeFile(params.path, params.content, 'utf8'); return { success: true };`

    - `fs/create`: If `params.type === 'directory'`: `mkdir(join(params.path, params.name), { recursive: false })`. Return `{ path: join(params.path, params.name), name: params.name, type: params.type }`.
    - `fs/rename`: `const newPath = join(dirname(params.oldPath), params.newName); await rename(params.oldPath, newPath); return { oldPath: params.oldPath, newPath, newName: params.newName };`
    - `fs/delete`: `const s = await stat(params.path); if (s.isDirectory()) await rm(params.path, { recursive: true, force: true }); else await unlink(params.path); return { success: true };`
    - `fs/exists`: `try { await access(params.path); return { exists: true }; } catch { return { exists: false }; }`
    - default: `return { error: { code: -32601, message: 'Method not implemented: ' + method } };`
